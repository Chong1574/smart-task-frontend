// Merge: cada item aqui trae __detail; buscar el normalized correspondiente por indice (pairing 1:1 con 'Drop unknown licenses').
const normalized = $('Drop unknown licenses').all();
// Strip HTML → plaintext para 'description'. El backend hace decode de entidades,
// aqui solo removemos tags. slice(0,2000) coincide con el limite del schema.
const stripHtml = (html) => String(html || '').replace(/<[^>]+>/g, ' ').replace(/\s{2,}/g, ' ').trim().slice(0, 2000);
const products = items.map((item, i) => {
  const n = normalized[i]?.json || {};
  const det = item?.json?.__detail || {};
  return {
    ...n,
    description: stripHtml(det.description),
    descriptionHtml: det.description || '',
    images: det.images || [],
    variants: det.variants || [],
    priceFrom: det.priceFrom ?? null,
    price: det.priceFrom ?? null,
  };
});
return [{ json: { products } }];
